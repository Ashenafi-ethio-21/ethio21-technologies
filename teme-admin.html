/** * Teme Astegni (ተሜ አስጠኚ) - 100% Free Cloud Database Engine * Primary Owner: ashenafitobe@gmail.com * Organization: Ethio 21 Technologies */ function doGet(e) { var action = (e && e.parameter) ? e.parameter.action : ''; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'getData') { var pSheet = ss.getSheetByName('Parents') || createParentsSheet(ss); var tSheet = ss.getSheetByName('Tutors') || createTutorsSheet(ss); var mSheet = ss.getSheetByName('Matches') || createMatchesSheet(ss); var parents = rowsToJson(pSheet); var tutors = rowsToJson(tSheet); var matches = rowsToJson(mSheet); var output = { parents: parents, tutors: tutors, matches: matches }; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'OK', message: 'Teme Astegni API Active' })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var postData = JSON.parse(e.postData.contents); var action = postData.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'addParent') { var sheet = ss.getSheetByName('Parents') || createParentsSheet(ss); var d = postData.data; sheet.appendRow([new Date(), d.name, d.phone, d.area, d.grade, d.genderPref, d.subjs, 'Pending']); } else if (action === 'addTutor') { var sheet = ss.getSheetByName('Tutors') || createTutorsSheet(ss); var d = postData.data; sheet.appendRow([new Date(), d.name, d.phone, d.gender, d.uni, d.year, d.cgpa, d.subj, d.govtId, 'Verified']); } else if (action === 'addMatch') { var sheet = ss.getSheetByName('Matches') || createMatchesSheet(ss); var d = postData.data; sheet.appendRow([new Date(), d.parentName, d.tutorName, d.monthlyFee, d.commission, d.status]); } return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function createParentsSheet(ss) { var s = ss.insertSheet('Parents'); s.appendRow(['Timestamp', 'Parent Name', 'Phone', 'Location', 'Grade', 'Gender Preference', 'Subjects', 'Status']); return s; } function createTutorsSheet(ss) { var s = ss.insertSheet('Tutors'); s.appendRow(['Timestamp', 'Full Name', 'Phone', 'Gender', 'University & Campus', 'Year', 'CGPA', 'Subjects', 'Government ID', 'Status']); return s; } function createMatchesSheet(ss) { var s = ss.insertSheet('Matches'); s.appendRow(['Timestamp', 'Parent Name', 'Tutor Name', 'Monthly Fee (ETB)', 'Commission (ETB)', 'Status']); return s; } function rowsToJson(sheet) { var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var headers = data[0]; var rows = []; for (var i = 1; i < data.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = data[i][j]; } rows.push(obj); } return rows; }
